# PR Inbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/pr-inbox` page that fetches PR review requests from a Slack channel, displays them with review status, and triggers AI code review on selected PRs.

**Architecture:** Server endpoint spawns Claude CLI to read Slack channel via MCP, then parses PR URLs server-side with regex. GitHub metadata and review status are enriched via `gh pr view` and the existing `PrReview` DB table. Frontend reuses `useRunnerJob` for review execution.

**Tech Stack:** Nuxt 4, Vue 3, TypeScript, Prisma (PrReview table), Claude CLI + Slack MCP, `gh` CLI

```ts
import { execSync } from 'node:child_process';
import process from 'node:process';

import { resolveClaudeCliPath } from '../../utils/claude-cli';
import prisma from '../../utils/prisma';
import { getAllRepos } from '../../utils/repo-mapping';
import { getPrInboxChannel } from '../../utils/workspaceConfig';

interface PrInboxItem {
  repo: string;
  prNumber: number;
  prTitle: string;
  prAuthor: string;
  htmlUrl: string;
  headSha: string;
  slackUser: string;
  slackTs: string;
  requestedAt: string;
  reviewStatus: 'closed' | 'not-reviewed' | 'outdated' | 'reviewed';
  repoLabel: string | null;
}

const PR_URL_RE = /https:\/\/github\.com\/([\w.-]+\/[\w.-]+)\/pull\/(\d+)/g;

/** Messages matching these patterns are NOT review requests */
const EXCLUSION_PATTERNS = [
  /PR Review Status/i,
  /該 code review 囉/,
];

interface SlackMessage {
  text: string;
  user: string;
  ts: string;
  subtype?: string;
}

function parseSlackOutput(raw: string): SlackMessage[] {
  // Claude CLI returns slack_read_channel output as formatted text.
  // Each message block starts with "=== Message from <user> ... at <date> ==="
  // and contains "Message TS: <ts>" followed by the message body.
  const messages: SlackMessage[] = [];
  const blocks = raw.split(/=== Message from /);

  for (const block of blocks) {
    if (!block.trim()) continue;

    const userMatch = /^(.+?)\s+\(([^)]+)\)\s+at\s+(.+?)\s*===/.exec(block);
    if (!userMatch) continue;

    const user = userMatch[1] ?? '';
    const tsMatch = /Message TS:\s*([\d.]+)/.exec(block);
    const ts = tsMatch?.[1] ?? '';

    // The message body is everything after the "Message TS: ..." line
    const bodyStart = block.indexOf('\n', block.indexOf('Message TS:'));
    const text = bodyStart >= 0 ? block.slice(bodyStart).trim() : '';

    // Detect bot subtype from the user ID format (B prefix = bot)
    const userId = userMatch[2] ?? '';
    const subtype = userId.startsWith('B') ? 'bot_message' : undefined;

    if (ts && text) {
      messages.push({ text, user, ts, subtype });
    }
  }

  return messages;
}

function shouldExclude(msg: SlackMessage): boolean {
  // Bot messages without PR URLs → exclude
  if (msg.subtype === 'bot_message' && !PR_URL_RE.test(msg.text)) {
    PR_URL_RE.lastIndex = 0;
    return true;
  }
  PR_URL_RE.lastIndex = 0;

  // Exclusion patterns
  for (const pattern of EXCLUSION_PATTERNS) {
    if (pattern.test(msg.text)) return true;
  }

  return false;
}

interface ParsedPr {
  repo: string;
  prNumber: number;
  htmlUrl: string;
  slackUser: string;
  slackTs: string;
  requestedAt: string;
}

function extractPrsFromMessages(messages: SlackMessage[]): ParsedPr[] {
  const seen = new Set<string>();
  const results: ParsedPr[] = [];

  for (const msg of messages) {
    if (shouldExclude(msg)) continue;

    // Reset regex state for each message
    PR_URL_RE.lastIndex = 0;
    let match: null | RegExpExecArray;
    while ((match = PR_URL_RE.exec(msg.text)) !== null) {
      const repo = match[1]!;
      const prNumber = Number(match[2]);
      const key = `${repo}#${prNumber}`;

      if (seen.has(key)) continue;
      seen.add(key);

      // Convert Slack ts to ISO date
      const unixSecs = Number.parseFloat(msg.ts);
      const requestedAt = new Date(unixSecs * 1000).toISOString();

      results.push({
        repo,
        prNumber,
        htmlUrl: match[0],
        slackUser: msg.user,
        slackTs: msg.ts,
        requestedAt,
      });
    }
  }

  return results;
}

function fetchPrMeta(
  repo: string,
  prNumber: number,
): { author: string; headSha: string; state: string; title: string } {
  try {
    const raw = execSync(
      `gh pr view ${prNumber} --repo ${repo} --json title,author,headRefOid,state`,
      { encoding: 'utf8', timeout: 10_000 },
    );
    const data = JSON.parse(raw) as {
      author: { login: string };
      headRefOid: string;
      state: string;
      title: string;
    };
    return {
      title: data.title,
      author: data.author.login,
      headSha: data.headRefOid,
      state: data.state,
    };
  } catch {
    return { title: `PR #${prNumber}`, author: 'unknown', headSha: '', state: 'OPEN' };
  }
}

export default defineEventHandler(async () => {
  const channel = await getPrInboxChannel();
  if (!channel) {
    throw createError({
      statusCode: 400,
      message: 'PR Inbox Slack channel not configured. Go to Settings → Integrations.',
    });
  }

  // 1. Fetch Slack messages via Claude CLI
  const twoDaysAgo = Math.floor(Date.now() / 1000) - 2 * 86400;
  const prompt = `Read Slack channel ${channel} messages from the last 2 days using slack_read_channel with oldest=${twoDaysAgo}. Print the full raw output exactly as returned. Do not summarize, filter, or reformat.`;

  const cliPath = resolveClaudeCliPath();
  const env: Record<string, string> = {
    ...Object.fromEntries(
      Object.entries(process.env).filter(
        (e): e is [string, string] => e[1] !== undefined,
      ),
    ),
    PATH: '/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin',
  };
  delete env.CLAUDECODE;
  delete env.CLAUDE_CODE_ENTRYPOINT;

  let slackRaw: string;
  try {
    slackRaw = execSync(
      `${cliPath} --dangerously-skip-permissions -p ${JSON.stringify(prompt)}`,
      { encoding: 'utf8', timeout: 60_000, env, cwd: process.cwd() },
    );
  } catch (error) {
    const msg = error instanceof Error ? error.message : String(error);
    throw createError({
      statusCode: 502,
      message: `Failed to read Slack channel: ${msg}`,
    });
  }

  // 2. Parse messages + extract PR URLs
  const messages = parseSlackOutput(slackRaw);
  const parsedPrs = extractPrsFromMessages(messages);

  // 3. Enrich with GitHub metadata
  const allRepos = await getAllRepos();
  const repoLabelMap = new Map(allRepos.map((r) => [r.githubRepo, r.label]));

  // 4. Build review status lookup from PrReview DB
  const existingReviews = await prisma.prReview.findMany({
    select: { repoLabel: true, prNumber: true, commitSha: true },
  });
  // Map: "repoLabel#prNumber" → Set<commitSha>
  const reviewMap = new Map<string, Set<string>>();
  for (const r of existingReviews) {
    const key = `${r.repoLabel}#${r.prNumber}`;
    let shas = reviewMap.get(key);
    if (!shas) {
      shas = new Set();
      reviewMap.set(key, shas);
    }
    shas.add(r.commitSha);
  }

  // 5. Enrich each PR
  const items: PrInboxItem[] = [];
  for (const pr of parsedPrs) {
    const meta = fetchPrMeta(pr.repo, pr.prNumber);
    const repoLabel = repoLabelMap.get(pr.repo) ?? null;

    let reviewStatus: PrInboxItem['reviewStatus'] = 'not-reviewed';
    if (meta.state === 'MERGED' || meta.state === 'CLOSED') {
      reviewStatus = 'closed';
    } else if (repoLabel) {
      const lookupKey = `${repoLabel}#${pr.prNumber}`;
      const shas = reviewMap.get(lookupKey);
      if (shas) {
        reviewStatus = shas.has(meta.headSha) ? 'reviewed' : 'outdated';
      }
    }

    items.push({
      repo: pr.repo,
      prNumber: pr.prNumber,
      prTitle: meta.title,
      prAuthor: meta.author,
      htmlUrl: pr.htmlUrl,
      headSha: meta.headSha,
      slackUser: pr.slackUser,
      slackTs: pr.slackTs,
      requestedAt: pr.requestedAt,
      reviewStatus,
      repoLabel,
    });
  }

  // Sort by requestedAt desc (newest first)
  items.sort((a, b) => b.requestedAt.localeCompare(a.requestedAt));

  return {
    items,
    fetchedAt: new Date().toISOString(),
    channel,
  };
});
```

```ts
import type { HistoryEntry } from '~/composables/useRunnerJob';

import { useRepoConfigs } from '~/composables/useRepoConfigs';

export interface PrInboxItem {
  repo: string;
  prNumber: number;
  prTitle: string;
  prAuthor: string;
  htmlUrl: string;
  headSha: string;
  slackUser: string;
  slackTs: string;
  requestedAt: string;
  reviewStatus: 'closed' | 'not-reviewed' | 'outdated' | 'reviewed';
  repoLabel: string | null;
}

type StatusFilter = 'all' | 'closed' | 'not-reviewed' | 'outdated' | 'reviewed';

function itemKey(item: PrInboxItem): string {
  return `${item.repo}#${item.prNumber}`;
}

export function usePrInbox() {
  const items = ref<PrInboxItem[]>([]);
  const loading = ref(false);
  const fetchError = ref('');
  const fetchedAt = ref<string | null>(null);
  const selected = ref<Set<string>>(new Set());
  const starting = ref(false);
  const statusFilter = ref<StatusFilter>('all');
  const history = ref<HistoryEntry[]>([]);
  const rightTab = ref<'history' | 'progress'>('progress');

  const { repoConfigs } = useRepoConfigs();

  const reviewer = useRunnerJob({
    storageKey: 'pr-inbox-active-jobId',
    apiBase: '/api/claude-runner',
    phases: [
      { label: '排隊中' },
      { label: '分析 PR' },
      { label: 'Review 中' },
      { label: '完成' },
    ],
    onComplete: () => {
      loadHistory();
    },
  });

  // Filtered list
  const filteredItems = computed(() => {
    if (statusFilter.value === 'all') return items.value;
    return items.value.filter((i) => i.reviewStatus === statusFilter.value);
  });

  // Group by slack user
  const groupedByUser = computed(() => {
    const map = new Map<string, PrInboxItem[]>();
    for (const item of filteredItems.value) {
      const arr = map.get(item.slackUser);
      if (arr) {
        arr.push(item);
      } else {
        map.set(item.slackUser, [item]);
      }
    }
    return map;
  });

  const selectedCount = computed(() => selected.value.size);

  // "N 分鐘前" display
  const fetchedAgo = computed(() => {
    if (!fetchedAt.value) return null;
    const diff = Date.now() - new Date(fetchedAt.value).getTime();
    const mins = Math.floor(diff / 60_000);
    if (mins < 1) return '剛剛';
    return `${mins} 分鐘前`;
  });

  async function fetchItems() {
    loading.value = true;
    fetchError.value = '';
    try {
      const data = await $fetch<{
        items: PrInboxItem[];
        fetchedAt: string;
        channel: string;
      }>('/api/pr-inbox/fetch', { method: 'POST' });
      items.value = data.items;
      fetchedAt.value = data.fetchedAt;
      selected.value = new Set();
    } catch (error) {
      const msg =
        (error as any)?.data?.message ||
        (error instanceof Error ? error.message : '讀取失敗');
      fetchError.value = msg;
    } finally {
      loading.value = false;
    }
  }

  function toggleItem(item: PrInboxItem) {
    if (reviewer.isRunning.value) return;
    if (!item.repoLabel) return; // can't review without repoLabel
    const key = itemKey(item);
    const next = new Set(selected.value);
    if (next.has(key)) {
      next.delete(key);
    } else {
      next.add(key);
    }
    selected.value = next;
  }

  function selectAllPending() {
    if (reviewer.isRunning.value) return;
    const next = new Set(selected.value);
    for (const item of filteredItems.value) {
      if (item.reviewStatus === 'not-reviewed' && item.repoLabel) {
        next.add(itemKey(item));
      }
    }
    selected.value = next;
  }

  function clearSelection() {
    selected.value = new Set();
  }

  async function runReview() {
    if (reviewer.isRunning.value || starting.value || selected.value.size === 0)
      return;

    // Group by repoLabel
    const byRepo = new Map<string, PrInboxItem[]>();
    for (const key of selected.value) {
      const item = items.value.find((i) => itemKey(i) === key);
      if (!item?.repoLabel) continue;
      const arr = byRepo.get(item.repoLabel);
      if (arr) {
        arr.push(item);
      } else {
        byRepo.set(item.repoLabel, [item]);
      }
    }

    if (byRepo.size === 0) return;

    starting.value = true;
    rightTab.value = 'progress';
    try {
      const allIssues: Array<{ key: string; summary: string }> = [];
      let firstJobId: string | undefined;

      for (const [repoLabel, prItems] of byRepo) {
        const result = await $fetch<{
          jobId?: string;
          message?: string;
          skipped?: boolean | string[];
        }>('/api/pr-review/run', {
          method: 'POST',
          body: {
            repoLabel,
            prNumbers: prItems.map((p) => p.prNumber),
          },
        });

        if (result.skipped === true) continue;

        if (result.jobId) {
          if (!firstJobId) firstJobId = result.jobId;
          const activePrs = prItems.filter((p) => {
            if (Array.isArray(result.skipped)) {
              return !result.skipped.includes(`#${p.prNumber}`);
            }
            return true;
          });
          allIssues.push(
            ...activePrs.map((p) => ({
              key: `#${p.prNumber}`,
              summary: `${repoLabel} — ${p.prTitle}`,
            })),
          );
        }
      }

      if (firstJobId && allIssues.length > 0) {
        reviewer.startJob(firstJobId, allIssues, undefined, undefined, 0);
        selected.value = new Set();
      } else {
        useToast().add({
          title: '已跳過',
          description: '所有選取的 PR 已經 Review 過',
          color: 'warning',
        });
      }
    } catch (error) {
      const msg =
        (error as any)?.data?.message ||
        (error instanceof Error ? error.message : '啟動失敗');
      useToast().add({ title: '啟動失敗', description: msg, color: 'error' });
    } finally {
      starting.value = false;
    }
  }

  async function loadHistory() {
    try {
      history.value = await $fetch<HistoryEntry[]>(
        '/api/claude-runner/jobs?type=pr-review',
      );
    } catch (error) {
      console.error('Failed to load PR inbox history:', error);
    }
  }

  return {
    items,
    loading,
    fetchError,
    fetchedAt,
    fetchedAgo,
    selected,
    selectedCount,
    starting,
    statusFilter,
    filteredItems,
    groupedByUser,
    history,
    rightTab,
    reviewer,
    fetchItems,
    toggleItem,
    selectAllPending,
    clearSelection,
    runReview,
    loadHistory,
  };
}
```

        <UIcon

      <div
        v-else-if="!inbox.loading.value && inbox.items.value.length === 0 && !inbox.fetchedAt.value"
        class="flex flex-1 flex-col items-center justify-center gap-3 text-[#444] select-none"

        <UIcon name="i-lucide-inbox" class="text-5xl" />
        <p class="text-sm text-[#444]">按「重新載入」從 Slack 抓取 PR</p>
      </div>

      <!-- PR list -->
      <div v-else class="flex-1 overflow-y-auto">
        <div
          v-for="[user, userItems] in inbox.groupedByUser.value"
          :key="user"
          class="border-b last:border-b-0"
          style="border-color: rgb(255 255 255 / 4%)"
        >
          <!-- User group header -->
          <div
            class="flex items-center gap-2 px-4 py-2"
            style="background: rgb(255 255 255 / 2%)"
          >
            <UIcon name="i-lucide-user" class="text-[#555]" />

            }"

          >
            <!-- Checkbox -->
            <div
              class="flex h-4 w-4 shrink-0 items-center justify-center rounded border"
              :class="
                isSelected(item.repo, item.prNumber)
                  ? 'border-[#8b5cf6] bg-[#8b5cf6]'
                  : 'border-[rgb(255_255_255/15%)]'
              "
            >
              <UIcon
                v-if="isSelected(item.repo, item.prNumber)"
                name="i-lucide-check"
                class="text-[10px] text-white"
              />
            </div>

            <!-- PR info -->
            <div class="min-w-0 flex-1">
              <div class="flex items-center gap-2">
                <a
                  :href="item.htmlUrl"
                  target="_blank"
                  rel="noopener"
                  class="shrink-0 font-mono text-xs font-semibold text-[#8b5cf6] underline-offset-2 hover:underline"
                  @click.stop
                >
                  #{{ item.prNumber }}
                </a>
                <span class="truncate text-xs text-[#ccc]">{{
                  item.prTitle
                }}</span>
              </div>
              <div class="mt-0.5 flex items-center gap-2 text-[10px] text-[#555]">
                <span
                  class="rounded bg-[rgb(255_255_255/4%)] px-1.5 py-0.5 font-mono"

                  {{ repoShort(item.repo) }}
                </span>
                <span>{{ item.prAuthor }}</span>
              </div>
            </div>

            <!-- Status badge -->
            <span
              class="shrink-0 rounded px-2 py-0.5 text-[10px] font-medium"
              :class="statusColor[item.reviewStatus]"
            >
              {{ statusLabel[item.reviewStatus] }}
            </span>

            <!-- No repo warning -->
            <UIcon
              v-if="!item.repoLabel"
              name="i-lucide-alert-triangle"
              class="shrink-0 text-yellow-500"
              title="Repo 未設定，無法 review"
            />
          </div>
        </div>
      </div>

      <!-- Bottom bar -->
      <div
        class="flex shrink-0 items-center gap-3 border-t px-4 py-3"
        style="border-color: rgb(255 255 255 / 6%)"
      >
        <button
          class="text-xs text-[#888] hover:text-[#ccc]"
          :disabled="inbox.reviewer.isRunning.value"
          @click="inbox.selectAllPending"
        >
          全選待 review
        </button>
        <button
          v-if="inbox.selectedCount.value > 0"
          class="text-xs text-[#555] hover:text-[#888]"
          @click="inbox.clearSelection"
        >
          取消全選
        </button>
        <div class="flex-1" />
        <button
          class="interactive rounded-lg px-4 py-2 text-sm font-medium text-[#fafafa] transition-colors"
          :class="
            inbox.selectedCount.value > 0 && !inbox.reviewer.isRunning.value
              ? 'bg-[#8b5cf6] hover:bg-[#7c3aed]'
              : 'cursor-not-allowed bg-[#333] text-[#555]'
          "
          :disabled="inbox.selectedCount.value === 0 || inbox.reviewer.isRunning.value || inbox.starting.value"
          @click="inbox.runReview"
        >
          {{
            inbox.starting.value
              ? '啟動中...'
              : `Review (${inbox.selectedCount.value})`
          }}
        </button>
      </div>
    </div>

    <!-- Right: execution panel -->
    <div class="flex flex-1 flex-col overflow-hidden">
      <!-- Tabs -->
      <div
        class="flex shrink-0 border-b"
        style="border-color: rgb(255 255 255 / 6%)"
      >
        <button
          class="px-4 py-2.5 text-xs font-medium transition-colors"
          :class="
            inbox.rightTab.value === 'progress'
              ? 'border-b-2 border-[#8b5cf6] text-[#fafafa]'
              : 'text-[#555] hover:text-[#888]'
          "
          @click="inbox.rightTab.value = 'progress'"
        >
          執行過程
        </button>
        <button
          class="flex items-center gap-1.5 px-4 py-2.5 text-xs font-medium transition-colors"
          :class="
            inbox.rightTab.value === 'history'
              ? 'border-b-2 border-[#8b5cf6] text-[#fafafa]'
              : 'text-[#555] hover:text-[#888]'
          "
          @click="inbox.rightTab.value = 'history'"
        >
          執行紀錄
          <span
            v-if="inbox.history.value.length > 0"
            class="rounded-full bg-[rgb(255_255_255/6%)] px-1.5 py-0.5 text-[9px]"
          >
            {{ inbox.history.value.length }}
          </span>
        </button>
      </div>

      <!-- Status row -->
      <RunnerStatusRow
        v-if="inbox.reviewer.activeJob.value"
        :active-job="inbox.reviewer.activeJob.value"
        :is-running="inbox.reviewer.isRunning.value"
        :success-count="inbox.reviewer.successCount.value"
        :error-count="inbox.reviewer.errorCount.value"

        :expanded="true"
        :hide-results="inbox.rightTab.value === 'progress'"
        @cancel="inbox.reviewer.cancelJob"
      />

      <!-- Progress -->
      <template v-if="inbox.rightTab.value === 'progress'">
        <RunnerJobProgress
          v-if="inbox.reviewer.activeJob.value"
          :active-job="inbox.reviewer.activeJob.value"

        />
        <div
          v-else
          class="flex flex-1 flex-col items-center justify-center gap-3 text-[#444] select-none"
        >
          <UIcon name="i-lucide-inbox" class="text-5xl" />
          <div class="text-center">
            <p class="font-medium text-[#444]">
              從左側勾選 PR，開始自動 Code Review
            </p>
          </div>

      </template>

      <!-- History -->
      <template v-else-if="inbox.rightTab.value === 'history'">
        <RunnerJobHistory :history="inbox.history.value" />
      </template>
    </div>
  </div>

<script setup lang="ts">
useHead({ title: 'Claude Runner — PR Inbox' });

type RunnerJob = ReturnType<
  typeof import('~/composables/useRunnerJob').useRunnerJob
>;

const inboxTab = ref<{
  fetchItems: () => Promise<void>;
  loadHistory: () => Promise<void>;
  reviewer: RunnerJob;
}>();

onMounted(async () => {
  inboxTab.value?.loadHistory();

  const rev = inboxTab.value?.reviewer;
  if (rev) {
    const jobId = localStorage.getItem(rev.storageKey);
    if (jobId) {
      await rev.restoreJob(jobId);
    }
  }
});

onBeforeUnmount(() => {
  inboxTab.value?.reviewer.cleanup();
});
</script>

<template>
  <div class="flex flex-1 overflow-hidden">
    <PrInboxTab ref="inboxTab" />
  </div>
</template>
